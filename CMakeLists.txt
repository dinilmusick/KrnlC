cmake_minimum_required(VERSION 3.16)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/version.txt" ULTRST_RELEASE_VERSION LIMIT_COUNT 1)
project(KrnlC VERSION ${ULTRST_RELEASE_VERSION} LANGUAGES C)
# Exact upstream compatibility pin. The freestanding C11 core does not consume GutC APIs.
set(KRNLC_GUTC_REFERENCE_SHA "0e26ac721c7a020078cfaf52e6fce259fe03301e" CACHE STRING "Reviewed GutC interface revision")
set(KRNLC_GUTC_SOURCE_DIR "" CACHE PATH "Optional pinned GutC checkout to verify; no GutC APIs currently consumed")
if(KRNLC_GUTC_SOURCE_DIR)
  execute_process(COMMAND git -C "${KRNLC_GUTC_SOURCE_DIR}" rev-parse HEAD OUTPUT_VARIABLE gutc_head OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE gutc_result)
  if(NOT gutc_result EQUAL 0 OR NOT gutc_head STREQUAL KRNLC_GUTC_REFERENCE_SHA)
    message(FATAL_ERROR "GutC checkout must match exact pinned SHA ${KRNLC_GUTC_REFERENCE_SHA}")
  endif()
endif()
add_library(krnlc STATIC src/lifecycle.c src/kernel.c)
add_library(KrnlC::krnlc ALIAS krnlc)
target_include_directories(krnlc PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
set_target_properties(krnlc PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
if(MSVC)
  target_compile_options(krnlc PRIVATE /W4 /WX)
else()
  target_compile_options(krnlc PRIVATE -Wall -Wextra -Werror -Wpedantic)
endif()
include(GNUInstallDirs)
install(TARGETS krnlc EXPORT KrnlCTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT KrnlCTargets FILE KrnlCTargets.cmake NAMESPACE KrnlC:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/KrnlC)
include(CMakePackageConfigHelpers)
configure_package_config_file(cmake/KrnlCConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/KrnlCConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/KrnlC)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/KrnlCConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/KrnlCConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/KrnlCConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/KrnlC)
include(CTest)
if(BUILD_TESTING)
 add_executable(krnlc_lifecycle_test tests/lifecycle_test.c)
 add_executable(krnlc_kernel_test tests/kernel_test.c)
 target_link_libraries(krnlc_lifecycle_test PRIVATE krnlc)
 target_link_libraries(krnlc_kernel_test PRIVATE krnlc)
 foreach(t krnlc_lifecycle_test krnlc_kernel_test)
  set_target_properties(${t} PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
  if(NOT MSVC)
    target_compile_options(${t} PRIVATE -Wall -Wextra -Werror -Wpedantic)
  endif()
 endforeach()
 add_test(NAME lifecycle COMMAND krnlc_lifecycle_test)
 add_test(NAME kernel COMMAND krnlc_kernel_test)
 add_test(NAME installed_consumer COMMAND ${CMAKE_COMMAND} -DPROJECT_BUILD=${CMAKE_BINARY_DIR} -DPROJECT_SOURCE=${CMAKE_SOURCE_DIR} -DCONFIG=$<CONFIG> "-DCONSUMER_C_FLAGS=${CMAKE_C_FLAGS}" "-DCONSUMER_LINKER_FLAGS=${CMAKE_EXE_LINKER_FLAGS}" -P ${CMAKE_SOURCE_DIR}/tests/installed_consumer.cmake)
 set_tests_properties(installed_consumer PROPERTIES RUN_SERIAL TRUE)
endif()
