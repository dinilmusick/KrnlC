cmake_minimum_required(VERSION 3.16)
project(KrnlC VERSION 0.1.0 LANGUAGES C)
add_library(krnlc STATIC src/lifecycle.c)
target_include_directories(krnlc PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
set_target_properties(krnlc PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
include(CTest)
if(BUILD_TESTING)
  add_executable(krnlc_lifecycle_test tests/lifecycle_test.c)
  target_link_libraries(krnlc_lifecycle_test PRIVATE krnlc)
  set_target_properties(krnlc_lifecycle_test PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
  add_test(NAME krnlc_lifecycle COMMAND krnlc_lifecycle_test)
endif()
